# Synthetic VDES signal generator using libliquid (exact RRC match with demod)
add_executable(vdes_synth_liq vdes_synth_liq.c)
set_property(TARGET vdes_synth_liq PROPERTY C_STANDARD 99)
target_link_libraries(vdes_synth_liq PRIVATE m)

# Link libliquid (same logic as plugins/CMakeLists.txt)
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
  pkg_check_modules(_LIQ QUIET IMPORTED_TARGET liquid)
endif()
if(_LIQ_FOUND)
  target_link_libraries(vdes_synth_liq PRIVATE PkgConfig::_LIQ)
else()
  find_library(_LIQLIQ NAMES liquid PATHS /usr/local/lib /usr/lib /usr/lib64)
  find_path(_LIQINC liquid/liquid.h PATHS /usr/local/include /usr/include)
  if(_LIQLIQ AND _LIQINC)
    target_include_directories(vdes_synth_liq PRIVATE ${_LIQINC})
    target_link_libraries(vdes_synth_liq PRIVATE ${_LIQLIQ})
  endif()
endif()
